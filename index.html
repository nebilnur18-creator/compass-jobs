function renderJobCard(job) { const categories = getCategories(job); const positions = getPositions(job); const expired = isExpired(job); const deadline = formatDeadline(job); const link = safeUrl(job.link); const title = getJobTitle(job); const company = job.company || "Unknown company"; const location = getLocation(job); const locationText = Array.isArray(location) ? location.join(", ") : String(location || ""); const tags = categories .slice(0, 4) .map(category => ` ${safeText(category)} `) .join(""); const positionsHtml = positions .slice(0, 6) .map(position => `
  • ${safeText(position)}
    • `) .join(""); const description = getShortDescription(job); /* STEP 2: Every job card now opens job.html first. Example: job.html?id=12345 The original application link remains available on the job details page. */ const jobId = String(job.id ?? "").trim(); const detailsUrl = jobId ? `job.html?id=${encodeURIComponent(jobId)}` : ""; const applyHtml = detailsUrl ? ` View details → ` : link ? ` View application ↗ ` : ` Job details unavailable `; return `
    ${safeText(company)}

    ${safeText(title)}

    ${tags} ${ locationText ? ` 📍 ${safeText(locationText)} ` : "" } ${ expired ? ` Closed ` : ` Active ` }
    ${ positionsHtml ? `
      ${positionsHtml}
    ` : "" } ${ description ? `
    ${safeText(description)}
    ` : "" }
    ${deadline.html}
    ${applyHtml}
    `; }
